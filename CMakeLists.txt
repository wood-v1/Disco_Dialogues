cmake_minimum_required(VERSION 3.20)
project(DiscoDialogues VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Disco Dialogues requires Windows Win32/x86.")
endif()
set(OYNONTOOLS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../OynonTools" CACHE PATH "OynonTools repository")
if(NOT EXISTS "${OYNONTOOLS_ROOT}/include/OynonRuntimeApi.h")
    message(FATAL_ERROR "Set OYNONTOOLS_ROOT to the OynonTools repository.")
endif()
include(CTest)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
add_subdirectory("${OYNONTOOLS_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/oynontools")
# Keep dependency artifacts in this build tree; package the DLL built with the mod.
set_target_properties(OynonTools PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
add_library(DiscoDialogues SHARED src/runtime.cpp src/dialog_camera.cpp src/dialog_feed.cpp src/dialog_speech.cpp)
target_compile_features(DiscoDialogues PRIVATE cxx_std_17)
target_compile_definitions(DiscoDialogues PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(DiscoDialogues PRIVATE OynonTools)
if(MSVC)
    target_compile_options(DiscoDialogues PRIVATE /W4 /WX /EHsc)
    set_property(TARGET DiscoDialogues PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(BUILD_TESTING)
    foreach(name camera_framing camera_transition)
        add_executable(${name}_test tests/${name}_test.cpp)
        target_compile_features(${name}_test PRIVATE cxx_std_17)
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
    # Integration fixture executes the real adapter and the mod policy together.
    foreach(name dialog_speech dialog_camera dialog_feed)
        add_executable(${name}_test tests/${name}_test.cpp)
        target_compile_features(${name}_test PRIVATE cxx_std_17)
        target_compile_definitions(${name}_test PRIVATE OYNONTOOLS_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
        target_include_directories(${name}_test PRIVATE "${OYNONTOOLS_ROOT}/include" "${OYNONTOOLS_ROOT}/src/runtime")
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
    set(INVENTORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../UtopianInventory" CACHE PATH "Inventory Overhaul repository")
    add_executable(inventory_dialog_input_test tests/inventory_dialog_input_test.cpp)
    target_compile_features(inventory_dialog_input_test PRIVATE cxx_std_17)
    target_include_directories(inventory_dialog_input_test PRIVATE "${INVENTORY_ROOT}" "${OYNONTOOLS_ROOT}/include" "${OYNONTOOLS_ROOT}/src")
    add_test(NAME inventory_dialog_input COMMAND inventory_dialog_input_test)
    add_executable(layout_selection_test tests/layout_selection_test.cpp)
    target_compile_features(layout_selection_test PRIVATE cxx_std_17)
    add_test(NAME layout_selection COMMAND layout_selection_test)
endif()
